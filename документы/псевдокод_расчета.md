# Псевдокод расчёта страниц и стоимости (из ЛР1)

```cpp
Начало:
input (file, copies, printType)
if (file не выбран || copies <= 0 || printType не указан) {
    printf "Заполните обязательные поля";
    end;
}
if (file.size > 20_MB) {
    printf "Файл больше 20 МБ";
    end;
}
if (file.format не в {PDF, DOCX, JPG, PNG}) {
    printf "Недопустимый формат файла";
    end;
}
if (file.format == PDF || file.format == DOCX) {
    pages = определитьЧислоСтраниц(file);
    if (pages не определены) {
        printf "Не удалось определить число страниц";
        end;
    }
} else {
    pages = 1;
}
if (printType == "ч/б") {
    price = 5;
} else {
    price = 12;
}
cost = pages * copies * price;
if (cost < 10) {
    cost = 10;
}
printf pages, copies, price, cost;
Конец;
